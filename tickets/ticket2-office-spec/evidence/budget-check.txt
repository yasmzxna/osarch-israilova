$rows = Import-Csv budget.csv -Encoding UTF8 | Where-Object { $_.role -ne 'ALL' -and $_.component -ne 'TOTAL' }
$rows | Group-Object role | ForEach-Object { [pscustomobject]@{ role = $_.Name; parts = $_.Count; sum_kzt = ($_.Group | Measure-Object price_kzt -Sum).Sum } } | Format-Table -AutoSize
$grand = ($rows | Measure-Object price_kzt -Sum).Sum

"Rows without a date: $(@($rows | Where-Object { -not $_.date }).Count)"
